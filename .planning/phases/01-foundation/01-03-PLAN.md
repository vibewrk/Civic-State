```dockerfile
# Production stage

RUN corepack enable && corepack prepare pnpm@latest --activate
RUN npm install -g pm2

WORKDIR /app
COPY --from=base /app/node_modules ./node_modules
COPY --from=base /app/packages/shared ./packages/shared
COPY --from=base /app/apps/api/dist ./apps/api/dist
COPY --from=base /app/apps/api/package.json ./apps/api/
COPY --from=base /app/apps/api/ecosystem.config.cjs ./apps/api/

WORKDIR /app/apps/api
CMD ["pm2-runtime", "ecosystem.config.cjs"]
```

1. Create `docker/nginx/nginx.conf` (per D-06):

```nginx
worker_processes auto;

events {
    worker_connections 1024;
}

http {
    # Security headers
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;

    # Rate limiting
    limit_req_zone $binary_remote_addr zone=api:10m rate=30r/s;

    # Upstream
    upstream api_server {
        server api:3001;
    }

    # HTTP -> HTTPS redirect (production)
    server {
        listen 80;
        server_name api.civicstate.com;

        location /.well-known/acme-challenge/ {
            root /var/lib/letsencrypt;
        }

        location / {
            return 301 https://$host$request_uri;
        }
    }

    # HTTPS server
    server {
        listen 443 ssl;
        server_name api.civicstate.com;

        ssl_certificate /etc/letsencrypt/live/api.civicstate.com/fullchain.pem;
        ssl_certificate_key /etc/letsencrypt/live/api.civicstate.com/privkey.pem;

        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers HIGH:!aNULL:!MD5;
        ssl_prefer_server_ciphers on;

        # CORS — only allow civicstate.com frontend (per D-06)
        # Note: CORS is primarily enforced by Express cors middleware.
        # Nginx adds headers for preflight when Express is unreachable.

        location /api/ {
            limit_req zone=api burst=50 nodelay;
            proxy_pass http://api_server;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            # Timeouts for long-running agent requests
            proxy_read_timeout 120s;
            proxy_connect_timeout 10s;
        }

        # Health check (no rate limit)
        location = /api/health {
            proxy_pass http://api_server;
            proxy_set_header Host $host;
        }
    }
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Internet -> Nginx | All external traffic enters through nginx (port 80/443) |
| Nginx -> API | Internal Docker network, trusted |
| Docker network | All inter-container communication on private network |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-07 | Denial of Service | Nginx | mitigate | Rate limiting: 30r/s per IP with burst=50, applied to /api/ routes |
| T-01-08 | Information Disclosure | Nginx | mitigate | Security headers (X-Frame-Options, X-Content-Type-Options, Referrer-Policy) |
| T-01-09 | Spoofing | SSL/TLS | mitigate | TLS 1.2+ only, Certbot auto-renewal, HTTPS redirect |
| T-01-10 | Tampering | Docker network | accept | Internal network is trusted; all containers on same host |
| T-01-11 | Information Disclosure | postgres port | mitigate | Production compose does NOT expose postgres to host; dev override does (acceptable for local dev) |
</threat_model>
