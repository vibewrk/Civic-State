```yaml
name: Deploy

on:
  push:
    branches: [main]

concurrency:
  group: deploy-production
  cancel-in-progress: false  # Never cancel active deploys

env:
  REGISTRY: ghcr.io
  API_IMAGE: ghcr.io/${{ github.repository_owner }}/civicstate-api
  WORKER_IMAGE: ghcr.io/${{ github.repository_owner }}/civicstate-worker

jobs:
  build-and-push:
    runs-on: ubuntu-latest
    permissions:
      contents: read
      packages: write

    steps:
      - uses: actions/checkout@v4

      - name: Log in to GHCR
        uses: docker/login-action@v3
        with:
          registry: ${{ env.REGISTRY }}
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - name: Build and push API image
        uses: docker/build-push-action@v6
        with:
          context: .
          file: docker/api.Dockerfile
          push: true
          tags: |
            ${{ env.API_IMAGE }}:${{ github.sha }}
            ${{ env.API_IMAGE }}:latest

      - name: Build and push Worker image
        uses: docker/build-push-action@v6
        with:
          context: .
          file: docker/worker.Dockerfile
          push: true
          tags: |
            ${{ env.WORKER_IMAGE }}:${{ github.sha }}
            ${{ env.WORKER_IMAGE }}:latest

  deploy:
    needs: build-and-push
    runs-on: ubuntu-latest
    steps:
      - name: Deploy to DigitalOcean droplet
        uses: appleboy/ssh-action@v1
        with:
          host: ${{ secrets.DROPLET_HOST }}
          username: ${{ secrets.DROPLET_USER }}
          key: ${{ secrets.DROPLET_SSH_KEY }}
          script: |
            cd /opt/civicstate
            export IMAGE_TAG=${{ github.sha }}
            docker compose -f docker/docker-compose.yml pull
            docker compose -f docker/docker-compose.yml up -d --remove-orphans

      - name: Health check
        run: |
          sleep 15
          for i in {1..10}; do
            STATUS=$(curl -sf https://api.civicstate.com/api/health | jq -r '.status' 2>/dev/null)
            if [ "$STATUS" = "ok" ]; then
              echo "Health check passed"
              exit 0
            fi
            echo "Attempt $i: status=$STATUS, retrying in 10s..."
            sleep 10
          done
          echo "Health check failed after 10 attempts"
          exit 1

      - name: Rollback on failure
        if: failure()
        uses: appleboy/ssh-action@v1
        with:
          host: ${{ secrets.DROPLET_HOST }}
          username: ${{ secrets.DROPLET_USER }}
          key: ${{ secrets.DROPLET_SSH_KEY }}
          script: |
            cd /opt/civicstate
            # Rollback to previous image (per D-08)
            export IMAGE_TAG=$(docker inspect --format='{{index .Config.Labels "org.opencontainers.image.revision"}}' civicstate-api 2>/dev/null || echo "latest")
            docker compose -f docker/docker-compose.yml up -d --force-recreate
```

1. Create `scripts/backup.sh` (per D-15, INFRA-08):

```bash
#!/usr/bin/env bash
set -euo pipefail

# CivicState PostgreSQL Backup Script
# Runs daily via cron on the DigitalOcean droplet
# Backs up to DigitalOcean Spaces (S3-compatible) with 30-day retention

TIMESTAMP=$(date +%Y%m%d_%H%M%S)
BACKUP_DIR="/tmp/civicstate-backups"
BACKUP_FILE="civicstate_${TIMESTAMP}.sql.gz"
SPACES_BUCKET="${DO_SPACES_BUCKET:-civicstate-backups}"
SPACES_REGION="${DO_SPACES_REGION:-nyc3}"
RETENTION_DAYS=30

# Create temp directory
mkdir -p "$BACKUP_DIR"

echo "[$(date)] Starting backup..."

# Dump database (from Docker container)
docker exec civicstate-postgres-1 pg_dump \
  -U civicstate \
  -d civicstate \
  --no-owner \
  --no-privileges \
  | gzip > "${BACKUP_DIR}/${BACKUP_FILE}"

BACKUP_SIZE=$(du -h "${BACKUP_DIR}/${BACKUP_FILE}" | cut -f1)
echo "[$(date)] Backup created: ${BACKUP_FILE} (${BACKUP_SIZE})"

# Upload to DigitalOcean Spaces via s3cmd
s3cmd put \
  "${BACKUP_DIR}/${BACKUP_FILE}" \
  "s3://${SPACES_BUCKET}/backups/${BACKUP_FILE}" \
  --host="${SPACES_REGION}.digitaloceanspaces.com" \
  --host-bucket="%(bucket)s.${SPACES_REGION}.digitaloceanspaces.com"

echo "[$(date)] Uploaded to Spaces: s3://${SPACES_BUCKET}/backups/${BACKUP_FILE}"

# Delete backups older than retention period
s3cmd ls "s3://${SPACES_BUCKET}/backups/" \
  --host="${SPACES_REGION}.digitaloceanspaces.com" \
  --host-bucket="%(bucket)s.${SPACES_REGION}.digitaloceanspaces.com" \
  | while read -r line; do
    FILE_DATE=$(echo "$line" | awk '{print $1}')
    FILE_PATH=$(echo "$line" | awk '{print $4}')
    if [[ -n "$FILE_DATE" && -n "$FILE_PATH" ]]; then
      DAYS_OLD=$(( ($(date +%s) - $(date -d "$FILE_DATE" +%s 2>/dev/null || date -j -f "%Y-%m-%d" "$FILE_DATE" +%s 2>/dev/null || echo 0)) / 86400 ))
      if [[ $DAYS_OLD -gt $RETENTION_DAYS ]]; then
        echo "[$(date)] Deleting old backup: $FILE_PATH ($DAYS_OLD days old)"
        s3cmd del "$FILE_PATH" \
          --host="${SPACES_REGION}.digitaloceanspaces.com" \
          --host-bucket="%(bucket)s.${SPACES_REGION}.digitaloceanspaces.com"
      fi
    fi
  done

# Cleanup local temp
rm -f "${BACKUP_DIR}/${BACKUP_FILE}"

echo "[$(date)] Backup complete."
```

2. Create `scripts/setup-dns.md` — DNS record setup for SPF/DKIM/DMARC (per D-24, DLVR-03):

Document the exact DNS TXT records needed on civicstate.com:

**SPF Record:**
- Type: TXT
- Host: @
- Value: `v=spf1 include:spf.mtasv.net ~all`
- (mtasv.net is Postmark's SPF include)

**DKIM Record:**
- Type: CNAME or TXT
- Host: provided by Postmark after domain verification
- Value: provided by Postmark (unique per domain)

**DMARC Record:**
- Type: TXT
- Host: _dmarc
- Value: `v=DMARC1; p=quarantine; rua=mailto:dmarc@civicstate.com; pct=100`

**Return-Path:**
- Type: CNAME
- Host: pm-bounces
- Value: `pm.mtasv.net`

3. Create `apps/api/src/routes/submissions.ts` — Test submission endpoint that triggers the full agent flow:

```typescript
import { Router } from 'express';
import { requireAuth } from '../middleware/auth.js';
import { z } from 'zod';

const router = Router();

const createSubmissionSchema = z.object({
  issueDescription: z.string().min(10).max(5000),
  desiredOutcome: z.string().min(10).max(2000),
  zipCode: z.string().regex(/^\d{5}(-\d{4})?$/),
  isAnonymous: z.boolean().default(true),
});

// POST /api/submissions — Create a new submission and trigger agent flow
// Public for Phase 1 testing; auth enforcement added at payment step
router.post('/api/submissions', async (req, res) => {
  try {
    const body = createSubmissionSchema.parse(req.body);

    const { prisma } = await import('shared');
    const { createSubmissionFlow } = await import('worker/src/engine/flow-producer.js').catch(() => {
      // Worker may not be importable from API in production
      // In production, use BullMQ Queue.add() directly
      return { createSubmissionFlow: null };
    });

    // Create submission in database
    const submission = await prisma.submission.create({
      data: {
        issueDescription: body.issueDescription,  // TODO: encrypt with CryptoService in Phase 2
        desiredOutcome: body.desiredOutcome,        // TODO: encrypt with CryptoService in Phase 2
        zipCode: body.zipCode,
        isAnonymous: body.isAnonymous,
        status: 'submitted',
        // userId will be set when auth is required
      },
    });

    // Create job record
    const job = await prisma.job.create({
      data: {
        submissionId: submission.id,
        type: 'submission',
        status: 'submitted',
        queue: 'submission',
      },
    });

    // Trigger agent flow via BullMQ (if worker is connected)
    // In production, this uses a Queue directly, not FlowProducer import
    const { Queue } = await import('bullmq');
    const { createRedisConnection } = await import('ioredis').then(() => {
      // Use inline connection for API->Queue communication
      return { createRedisConnection: () => {} };
    }).catch(() => ({ createRedisConnection: () => {} }));

    try {
      const submissionQueue = new Queue('classifier', {
        connection: {
          host: new URL(process.env.REDIS_URL || 'redis://localhost:6379').hostname,
          port: parseInt(new URL(process.env.REDIS_URL || 'redis://localhost:6379').port || '6379'),
        },
      });

      await submissionQueue.add(`classify-${submission.id}`, {
        submissionId: submission.id,
        jobId: job.id,
        ...body,
      });

      await submissionQueue.close();
    } catch (queueErr) {
      console.warn('Could not enqueue job (worker may not be running):', queueErr);
    }

    res.status(201).json({
      id: submission.id,
      jobId: job.id,
      status: 'submitted',
    });
  } catch (err) {
    if (err instanceof z.ZodError) {
      return res.status(400).json({ error: 'Validation failed', details: err.errors });
    }
    throw err;
  }
});

// GET /api/submissions/:id/status — Check job status
router.get('/api/submissions/:id/status', async (req, res) => {
  const { prisma } = await import('shared');

  const job = await prisma.job.findFirst({
    where: { submissionId: req.params.id },
    orderBy: { createdAt: 'desc' },
  });

  if (!job) {
    return res.status(404).json({ error: 'Submission not found' });
  }

  res.json({
    submissionId: req.params.id,
    jobId: job.id,
    status: job.status,
    createdAt: job.createdAt,
    completedAt: job.completedAt,
  });
});

export default router;
```
