```typescript
import { Router } from 'express';
import Stripe from 'stripe';
import { getAuth } from '@clerk/express';

const router: Router = Router();

// Pricing tiers (PAY-01) — 40% net margin floor enforced (PAY-04)
// $5 tier: $5 - $0.45 Stripe fee = $4.55 revenue. Cost ~$0.20 AI = 96% margin ✓
// $15 tier: $15 - $0.74 Stripe fee = $14.26 revenue. Cost ~$0.40 AI = 97% margin ✓
// $25 tier: $25 - $1.03 Stripe fee = $23.97 revenue. Cost ~$0.60 AI = 97% margin ✓
const PRICING_TIERS = {
  single: { price: 500, label: '1 Official', officialCount: 1 },       // $5.00
  three_pack: { price: 1500, label: '3 Officials', officialCount: 3 }, // $15.00
  full_spread: { price: 2500, label: 'All Officials', officialCount: -1 }, // $25.00 (-1 = all)
} as const;

function getStripe(): Stripe {
  return new Stripe(process.env.STRIPE_SECRET_KEY || '', {
    apiVersion: '2025-03-31.basil',
  });
}

// POST /api/submissions/:id/pay — Create Stripe Checkout session (PAY-02)
router.post('/:id/pay', async (req, res) => {
  const { userId } = getAuth(req);
  if (!userId) {
    return res.status(401).json({ error: { code: 'UNAUTHORIZED', message: 'Authentication required for payment' } });
  }
  
  const { id } = req.params;
  const { pricingTier } = req.body;
  
  if (!pricingTier || !PRICING_TIERS[pricingTier as keyof typeof PRICING_TIERS]) {
    return res.status(400).json({ error: { code: 'INVALID_TIER', message: 'Invalid pricing tier' } });
  }
  
  const tier = PRICING_TIERS[pricingTier as keyof typeof PRICING_TIERS];
  const stripe = getStripe();
  
  try {
    const { prisma } = await import('shared');
    
    // Find campaign for this submission
    const campaign = await prisma.campaign.findFirst({
      where: { submissionId: id },
    });
    
    if (!campaign) {
      return res.status(404).json({ error: { code: 'NOT_FOUND', message: 'No campaign found' } });
    }
    
    // Update campaign pricing
    await prisma.campaign.update({
      where: { id: campaign.id },
      data: { pricingTier, officialCount: tier.officialCount === -1 ? campaign.officialCount : tier.officialCount },
    });
    
    // Create Stripe Checkout session
    const session = await stripe.checkout.sessions.create({
      payment_method_types: ['card'],
      line_items: [{
        price_data: {
          currency: 'usd',
          product_data: {
            name: `CivicState Letter - ${tier.label}`,
            description: `Civic advocacy letter delivered to ${tier.label.toLowerCase()}`,
          },
          unit_amount: tier.price,
        },
        quantity: 1,
      }],
      mode: 'payment',
      success_url: `${process.env.FRONTEND_URL || 'http://localhost:3000'}/submit/success?session_id={CHECKOUT_SESSION_ID}`,
      cancel_url: `${process.env.FRONTEND_URL || 'http://localhost:3000'}/submit/cancel`,
      metadata: {
        submissionId: id,
        campaignId: campaign.id,
        pricingTier,
      },
    });
    
    // Create payment record
    await prisma.payment.create({
      data: {
        campaignId: campaign.id,
        userId: campaign.userId,
        stripeSessionId: session.id,
        amount: tier.price,
        currency: 'usd',
        status: 'pending',
      },
    });
    
    res.json({ checkoutUrl: session.url, sessionId: session.id });
  } catch (error) {
    console.error('Stripe Checkout error:', error);
    res.status(500).json({ error: { code: 'PAYMENT_ERROR', message: 'Failed to create checkout session' } });
  }
});

export default router;
```

1. Create `apps/api/src/routes/webhooks.ts`:

```typescript
import { Router, raw } from 'express';
import Stripe from 'stripe';

const router: Router = Router();

function getStripe(): Stripe {
  return new Stripe(process.env.STRIPE_SECRET_KEY || '', {
    apiVersion: '2025-03-31.basil',
  });
}

// POST /api/webhooks/stripe — Process payment confirmations (PAY-03)
// CRITICAL: raw body required for signature verification
router.post('/stripe', raw({ type: 'application/json' }), async (req, res) => {
  const stripe = getStripe();
  const sig = req.headers['stripe-signature'];
  const webhookSecret = process.env.STRIPE_WEBHOOK_SECRET;
  
  if (!sig || !webhookSecret) {
    return res.status(400).json({ error: 'Missing signature or webhook secret' });
  }
  
  let event: Stripe.Event;
  try {
    event = stripe.webhooks.constructEvent(req.body, sig, webhookSecret);
  } catch (err) {
    console.error('Webhook signature verification failed:', err);
    return res.status(400).json({ error: 'Invalid signature' });
  }
  
  if (event.type === 'checkout.session.completed') {
    const session = event.data.object as Stripe.Checkout.Session;
    const { submissionId, campaignId, pricingTier } = session.metadata || {};
    
    if (!submissionId || !campaignId) {
      console.error('Missing metadata in Stripe session');
      return res.status(200).json({ received: true });
    }
    
    try {
      const { prisma } = await import('shared');
      const { computeRowHmac } = await import('shared/hmac');
      
      // Update payment status
      await prisma.payment.updateMany({
        where: { stripeSessionId: session.id },
        data: {
          status: 'completed',
          stripePaymentIntentId: session.payment_intent as string,
        },
      });
      
      // Update campaign status to paid
      await prisma.campaign.update({
        where: { id: campaignId },
        data: { status: 'paid' },
      });
      
      // Transition job to paid state
      const job = await prisma.job.findFirst({
        where: { submissionId },
        orderBy: { createdAt: 'desc' },
      });
      
      if (job && job.status === 'payment_pending') {
        await prisma.job.update({
          where: { id: job.id },
          data: { status: 'paid' },
        });
        
        // Enqueue delivery job
        const { Queue } = await import('bullmq');
        const { createRedisConnection } = await import('../../worker/src/engine/connection.js');
        const deliveryQueue = new Queue('delivery', { connection: createRedisConnection() });
        await deliveryQueue.add(`deliver-${submissionId}`, {
          submissionId,
          campaignId,
          pricingTier,
        });
        await deliveryQueue.close();
      }
      
      // Log to audit trail
      const auditData = {
        userId: session.customer as string || '',
        action: 'campaign.paid' as const,
        resource: 'campaign',
        resourceId: campaignId,
        details: { amount: session.amount_total, pricingTier, stripeSessionId: session.id },
        createdAt: new Date(),
      };
      const hmac = computeRowHmac(auditData);
      await prisma.auditLog.create({ data: { ...auditData, details: auditData.details as any, hmacChecksum: hmac } });
      
    } catch (error) {
      console.error('Webhook processing error:', error);
      return res.status(500).json({ error: 'Processing failed' });
    }
  }
  
  res.status(200).json({ received: true });
});

export default router;
```
