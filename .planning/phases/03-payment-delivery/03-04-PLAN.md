```typescript
export async function createPaymentSession(submissionId: string, pricingTier: string): Promise<{ checkoutUrl: string }> {
  const res = await fetch(`${API_URL}/api/submissions/${submissionId}/pay`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    credentials: 'include',
    body: JSON.stringify({ pricingTier }),
  });
  if (!res.ok) {
    const error = await res.json();
    throw new Error(error.error?.message || 'Payment failed');
  }
  return res.json();
}
```
