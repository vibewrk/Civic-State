import * as postmark from 'postmark';

let client: postmark.ServerClient | null = null;

export function getPostmarkClient(): postmark.ServerClient {
  if (!client) {
    client = new postmark.ServerClient(process.env.POSTMARK_SERVER_TOKEN || '');
  }
  return client;
}

// Check per-domain bounce rate (DLVR-05)
export async function getDomainBounceRate(domain: string): Promise<number> {
  const { prisma } = await import('shared');
  
  const thirtyDaysAgo = new Date(Date.now() - 30 * 24 * 60 * 60 * 1000);
  
  const deliveries = await prisma.delivery.findMany({
    where: {
      letter: {
        official: {
          email: { endsWith: `@${domain}` },
        },
      },
      createdAt: { gte: thirtyDaysAgo },
    },
  });
  
  if (deliveries.length === 0) return 0;
  
  const bounced = deliveries.filter(d => d.status === 'bounced').length;
  return bounced / deliveries.length;
}

// Send individual email via Postmark (DLVR-01, DLVR-02)
export async function sendLetterEmail(params: {
  to: string;
  from: string;
  replyTo: string;
  subject: string;
  htmlBody: string;
  textBody: string;
  tag: string;
  metadata: Record<string, string>;
}): Promise<{ messageId: string }> {
  const pm = getPostmarkClient();
  
  const result = await pm.sendEmail({
    From: params.from,
    To: params.to,
    ReplyTo: params.replyTo,
    Subject: params.subject,
    HtmlBody: params.htmlBody,
    TextBody: params.textBody,
    Tag: params.tag,
    TrackOpens: true,
    TrackLinks: 'HtmlAndText' as any,
    Metadata: params.metadata,
    MessageStream: 'outbound',
  });
  
  return { messageId: result.MessageID };
}

import { Worker, Job } from 'bullmq';
import { createRedisConnection } from '../engine/connection.js';
import { getAgentConfig } from '../engine/config.js';
import { transitionJob } from '../engine/state-machine.js';
import { logAgentAction } from '../lib/logger.js';

const config = getAgentConfig('delivery');

async function processJob(job: Job): Promise<void> {
  const { submissionId, campaignId } = job.data;
  const startTime = Date.now();
  
  // Transition to delivering
  await transitionJob(submissionId, 'paid', 'delivering', config.name);
  
  try {
    const { prisma } = await import('shared');
    
    // Get campaign with letters and officials
    const campaign = await prisma.campaign.findUnique({
      where: { id: campaignId },
      include: {
        letters: {
          include: { official: true },
          where: { deletedAt: null },
        },
        submission: true,
      },
    });
    
    if (!campaign) throw new Error(`Campaign ${campaignId} not found`);
    
    let sentCount = 0;
    let failedCount = 0;
    
    for (const letter of campaign.letters) {
      const official = letter.official;
      
      // Skip opted-out officials (DLVR-07 downstream effect)
      if (official.optedOut) continue;
      
      // Skip officials without email
      if (!official.email || official.email.includes('@placeholder')) continue;
      
      // Check per-domain bounce rate (DLVR-05)
      const domain = official.email.split('@')[1];
      // Inline bounce rate check
      const thirtyDaysAgo = new Date(Date.now() - 30 * 24 * 60 * 60 * 1000);
      const domainDeliveries = await prisma.delivery.count({
        where: {
          letter: { official: { email: { endsWith: `@${domain}` } } },
          createdAt: { gte: thirtyDaysAgo },
        },
      });
      const domainBounces = await prisma.delivery.count({
        where: {
          letter: { official: { email: { endsWith: `@${domain}` } } },
          status: 'bounced',
          createdAt: { gte: thirtyDaysAgo },
        },
      });
      
      if (domainDeliveries > 0 && (domainBounces / domainDeliveries) > 0.10) {
        console.warn(`Skipping ${domain}: bounce rate ${(domainBounces/domainDeliveries*100).toFixed(1)}% exceeds 10% threshold`);
        await prisma.delivery.create({
          data: { letterId: letter.id, status: 'failed', postmarkMessageId: '' },
        });
        failedCount++;
        continue;
      }
      
      try {
        // Send via Postmark (DLVR-01, DLVR-02)
        const postmark = await import('postmark');
        const pmClient = new postmark.ServerClient(process.env.POSTMARK_SERVER_TOKEN || '');
        
        const replyTo = `reply+${campaignId}@civicstate.com`; // DLVR-08
        
        const result = await pmClient.sendEmail({
          From: 'letters@civicstate.com',
          To: official.email,
          ReplyTo: replyTo,
          Subject: `Constituent Communication Regarding ${campaign.submission?.zipCode || 'Your District'}`,
          TextBody: letter.content,
          HtmlBody: `<div style="font-family: Georgia, serif; max-width: 600px; margin: 0 auto;">${letter.content.replace(/\n/g, '<br>')}</div>`,
          Tag: `campaign-${campaignId}`,
          TrackOpens: true,
          MessageStream: 'outbound',
          Metadata: {
            campaignId,
            letterId: letter.id,
            officialId: official.id,
          },
        });
        
        // Create delivery record (DLVR-06)
        await prisma.delivery.create({
          data: {
            letterId: letter.id,
            postmarkMessageId: result.MessageID,
            status: 'sent',
            sentAt: new Date(),
          },
        });
        
        // Update letter status
        await prisma.letter.update({
          where: { id: letter.id },
          data: { status: 'sent' },
        });
        
        sentCount++;
      } catch (error) {
        console.error(`Delivery failed for letter ${letter.id}:`, error);
        await prisma.delivery.create({
          data: { letterId: letter.id, status: 'failed', postmarkMessageId: '' },
        });
        failedCount++;
      }
    }
    
    // Log agent action
    await logAgentAction({
      jobId: submissionId,
      agent: config.name,
      action: 'deliver',
      result: { sentCount, failedCount, totalLetters: campaign.letters.length },
      modelUsed: config.model,
      inputTokens: 0,
      outputTokens: 0,
      durationMs: Date.now() - startTime,
    });
    
    // Update campaign status
    await prisma.campaign.update({
      where: { id: campaignId },
      data: { status: failedCount === campaign.letters.length ? 'failed' : 'delivered' },
    });
    
    // Transition to delivered
    await transitionJob(submissionId, 'delivering', 'delivered', config.name);
  } catch (error) {
    console.error(`Delivery agent failed for ${submissionId}:`, error);
    await transitionJob(submissionId, 'delivering', 'failed', config.name);
  }
}

export const deliveryWorker = new Worker(config.queue, processJob, {
  connection: createRedisConnection(),
  concurrency: 5,
});

deliveryWorker.on('failed', (job, err) => {
  console.error(`Delivery failed for job ${job?.id}:`, err.message);
});

// POST /api/webhooks/postmark — Delivery event tracking (DLVR-06, DLVR-07)
router.post('/postmark', express.json(), async (req, res) => {
  const event = req.body;
  
  try {
    const { prisma } = await import('shared');
    const { computeRowHmac } = await import('shared/hmac');
    
    const messageId = event.MessageID;
    if (!messageId) return res.status(200).json({ received: true });
    
    const delivery = await prisma.delivery.findFirst({
      where: { postmarkMessageId: messageId },
      include: { letter: { include: { official: true } } },
    });
    
    if (!delivery) return res.status(200).json({ received: true });
    
    switch (event.RecordType) {
      case 'Delivery':
        await prisma.delivery.update({
          where: { id: delivery.id },
          data: { status: 'delivered', deliveredAt: new Date(event.DeliveredAt) },
        });
        break;
        
      case 'Bounce':
        await prisma.delivery.update({
          where: { id: delivery.id },
          data: { status: 'bounced', bouncedAt: new Date(), bounceType: event.Type || 'unknown' },
        });
        // Increment official bounce count
        await prisma.official.update({
          where: { id: delivery.letter.officialId },
          data: { bounceCount: { increment: 1 } },
        });
        break;
        
      case 'SpamComplaint':
        // DLVR-07: Immediate official suppression + operator alert
        await prisma.delivery.update({
          where: { id: delivery.id },
          data: { status: 'bounced', spamComplaintAt: new Date() },
        });
        await prisma.official.update({
          where: { id: delivery.letter.officialId },
          data: { optedOut: true },
        });
        
        // Log to audit trail
        const auditData = {
          action: 'official.suppressed' as const,
          resource: 'official',
          resourceId: delivery.letter.officialId,
          details: { reason: 'spam_complaint', messageId, officialEmail: delivery.letter.official.email },
          createdAt: new Date(),
        };
        const hmac = computeRowHmac(auditData);
        await prisma.auditLog.create({ data: { ...auditData, details: auditData.details as any, hmacChecksum: hmac } });
        
        // TODO: Send operator alert email
        console.error(`⚠️ SPAM COMPLAINT: Official ${delivery.letter.official.name} (${delivery.letter.official.email}) suppressed`);
        break;
    }
  } catch (error) {
    console.error('Postmark webhook error:', error);
  }
  
  res.status(200).json({ received: true });
});
