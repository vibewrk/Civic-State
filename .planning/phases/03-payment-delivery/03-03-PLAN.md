import { computeRowHmac } from 'shared/hmac';

// Estimated cost per submission by tier
const ESTIMATED_COSTS = {
  single: 20,       // $0.20 in cents
  three_pack: 40,   // $0.40
  full_spread: 60,  // $0.60
};

// Record payment to append-only ledger (TRSY-01)
export async function recordLedgerEntry(params: {
  type: 'payment' | 'refund' | 'api_cost' | 'postage' | 'adjustment';
  amount: number; // in cents
  reference: string;
  description: string;
  jobId?: string;
}): Promise<void> {
  const { prisma } = await import('shared');
  
  const data = {
    type: params.type,
    amount: params.amount,
    currency: 'usd',
    reference: params.reference,
    description: params.description,
    jobId: params.jobId || null,
    createdAt: new Date(),
  };
  
  const hmac = computeRowHmac(data);
  await prisma.ledgerEntry.create({ data: { ...data, hmacChecksum: hmac } });
}

// Job-level budget authorization (TRSY-02)
export async function authorizeJobBudget(jobId: string, pricingTier: string): Promise<{
  authorized: boolean;
  estimatedCost: number;
  ceiling: number;
}> {
  const estimated = ESTIMATED_COSTS[pricingTier as keyof typeof ESTIMATED_COSTS] || 60;
  const ceiling = Math.ceil(estimated * 1.5); // 150% ceiling (TRSY-03)
  
  return { authorized: true, estimatedCost: estimated, ceiling };
}

// Check if job has exceeded budget ceiling (TRSY-03)
export async function checkBudgetCeiling(jobId: string, pricingTier: string): Promise<{
  exceeded: boolean;
  actualCost: number;
  ceiling: number;
}> {
  const { prisma } = await import('shared');
  
  const costs = await prisma.ledgerEntry.aggregate({
    where: { jobId, type: 'api_cost' },
    _sum: { amount: true },
  });
  
  const actualCost = costs._sum.amount || 0;
  const estimated = ESTIMATED_COSTS[pricingTier as keyof typeof ESTIMATED_COSTS] || 60;
  const ceiling = Math.ceil(estimated * 1.5);
  
  return { exceeded: actualCost > ceiling, actualCost, ceiling };
}

// Daily reconciliation (TRSY-04, TRSY-05)
export async function runDailyReconciliation(): Promise<{
  totalRevenue: number;
  totalCosts: number;
  netProfit: number;
  discrepancies: Array<{ type: string; amount: number; details: string }>;
}> {
  const { prisma } = await import('shared');
  
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const tomorrow = new Date(today);
  tomorrow.setDate(tomorrow.getDate() + 1);
  
  const entries = await prisma.ledgerEntry.findMany({
    where: { createdAt: { gte: today, lt: tomorrow } },
  });
  
  const revenue = entries.filter(e => e.type === 'payment').reduce((sum, e) => sum + e.amount, 0);
  const costs = entries.filter(e => ['api_cost', 'postage'].includes(e.type)).reduce((sum, e) => sum + e.amount, 0);
  const refunds = entries.filter(e => e.type === 'refund').reduce((sum, e) => sum + e.amount, 0);
  
  // Check for discrepancies (TRSY-05)
  const discrepancies: Array<{ type: string; amount: number; details: string }> = [];
  
  // Verify payment records match Stripe (simplified — full Stripe reconciliation in production)
  const payments = await prisma.payment.findMany({
    where: { createdAt: { gte: today, lt: tomorrow }, status: 'completed' },
  });
  
  const stripeTotal = payments.reduce((sum, p) => sum + p.amount, 0);
  const ledgerPaymentTotal = revenue;
  
  if (Math.abs(stripeTotal - ledgerPaymentTotal) > 10) { // >$0.10 threshold
    discrepancies.push({
      type: 'payment_mismatch',
      amount: stripeTotal - ledgerPaymentTotal,
      details: `Stripe total: ${stripeTotal}, Ledger total: ${ledgerPaymentTotal}`,
    });
  }
  
  return {
    totalRevenue: revenue - refunds,
    totalCosts: costs,
    netProfit: (revenue - refunds) - costs,
    discrepancies,
  };
}

// Generate daily P&L report (TRSY-07)
export function formatPnLEmail(reconciliation: Awaited<ReturnType<typeof runDailyReconciliation>>): {
  subject: string;
  body: string;
} {
  const date = new Date().toLocaleDateString('en-US', { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' });
  
  return {
    subject: `CivicState Daily P&L — ${date}`,
    body: `
CivicState Daily Financial Report
${date}

Revenue:  $${(reconciliation.totalRevenue / 100).toFixed(2)}
Costs:    $${(reconciliation.totalCosts / 100).toFixed(2)}
Net P&L:  $${(reconciliation.netProfit / 100).toFixed(2)}

${reconciliation.discrepancies.length > 0
  ? `⚠️ DISCREPANCIES FOUND:\n${reconciliation.discrepancies.map(d => `  - ${d.type}: $${(d.amount / 100).toFixed(2)} (${d.details})`).join('\n')}`
  : '✅ No discrepancies found.'}

---
Automated report from CivicState Treasury
    `.trim(),
  };
}
