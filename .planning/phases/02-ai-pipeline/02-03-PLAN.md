```typescript
import Anthropic from '@anthropic-ai/sdk';
import { getAgentConfig } from '../engine/config.js';
import { logAgentAction } from './logger.js';

let client: Anthropic | null = null;

export function getAnthropicClient(): Anthropic {
  if (!client) {
    client = new Anthropic();
  }
  return client;
}

export interface LLMCallResult {
  text: string;
  inputTokens: number;
  outputTokens: number;
  cacheReadTokens: number;
  cacheCreationTokens: number;
}

// Call Anthropic API with automatic token logging
export async function callWithLogging(
  agentName: string,
  jobId: string,
  action: string,
  systemPrompt: string,
  userMessage: string,
  options?: { maxTokens?: number; temperature?: number },
): Promise<LLMCallResult> {
  const config = getAgentConfig(agentName);
  const anthropic = getAnthropicClient();
  const startTime = Date.now();
  
  const response = await anthropic.messages.create({
    model: config.model,
    max_tokens: options?.maxTokens || config.maxTokens,
    temperature: options?.temperature ?? 0.3,
    system: [
      {
        type: 'text',
        text: systemPrompt,
        cache_control: config.systemPromptCacheKey ? { type: 'ephemeral' } : undefined,
      },
    ],
    messages: [{ role: 'user', content: userMessage }],
  });
  
  const text = response.content[0]?.type === 'text' ? response.content[0].text : '';
  const usage = response.usage;
  
  const result: LLMCallResult = {
    text,
    inputTokens: usage.input_tokens,
    outputTokens: usage.output_tokens,
    cacheReadTokens: (usage as any).cache_read_input_tokens || 0,
    cacheCreationTokens: (usage as any).cache_creation_input_tokens || 0,
  };
  
  // Log token usage (AGNT-05)
  await logAgentAction({
    jobId,
    agent: config.name,
    action,
    result: {
      inputTokens: result.inputTokens,
      outputTokens: result.outputTokens,
      cacheReadTokens: result.cacheReadTokens,
      cacheCreationTokens: result.cacheCreationTokens,
    },
    modelUsed: config.model,
    inputTokens: result.inputTokens,
    outputTokens: result.outputTokens,
    durationMs: Date.now() - startTime,
  });
  
  return result;
}
```

```typescript
import { Worker, Job } from 'bullmq';
import { createRedisConnection } from '../engine/connection.js';
import { getAgentConfig } from '../engine/config.js';
import { transitionJob } from '../engine/state-machine.js';
import { callWithLogging } from '../lib/anthropic.js';

const config = getAgentConfig('classifier');

const CLASSIFIER_SYSTEM_PROMPT = `You are a civic issue classifier for CivicState, a platform that helps citizens write letters to government officials.

Analyze the user's civic issue and classify it. Respond with ONLY valid JSON:

{
  "issueType": "policy" | "enforcement" | "legislation" | "budget" | "service" | "other",
  "jurisdictionLevel": "federal" | "state" | "local" | "multiple",
  "severity": "low" | "medium" | "high" | "critical",
  "confidence": 0.0-1.0,
  "issueCategories": ["housing", "environment", "education", "public_safety", "transportation", "healthcare", "taxes", "civil_rights", "infrastructure", "other"],
  "summary": "One-sentence summary of the issue for internal use"
}

Issue types:
- policy: Requesting change to existing policy or rule
- enforcement: Requesting enforcement of existing law/regulation
- legislation: Requesting new law or amendment
- budget: Related to funding/spending decisions
- service: Government service quality issue
- other: Doesn't fit above categories

Jurisdiction guidance:
- federal: EPA, FDA, FCC, immigration, military, Social Security, Medicare, federal taxes
- state: Education standards, state highways, professional licensing, state taxes, criminal law
- local: Zoning, utilities, trash, roads, parks, police, fire, building permits
- multiple: Issues spanning multiple levels (e.g., housing policy involves all three)`;

async function processJob(job: Job): Promise<void> {
  const { submissionId, issueDescription, desiredOutcome } = job.data;
  
  // Transition to classifying
  await transitionJob(submissionId, 'submitted', 'classifying', config.name);
  
  try {
    const userMessage = `Issue: ${issueDescription}\n\nDesired Outcome: ${desiredOutcome}`;
    
    const llmResult = await callWithLogging(
      'classifier',
      submissionId,
      'classify',
      CLASSIFIER_SYSTEM_PROMPT,
      userMessage,
      { temperature: 0.1 }, // Low temperature for classification
    );
    
    const classification = JSON.parse(llmResult.text);
    
    // Store classification result on the job for downstream agents
    await job.updateData({
      ...job.data,
      classification,
    });
    
    // Update submission with classification result
    const { prisma } = await import('shared');
    await prisma.submission.update({
      where: { id: submissionId },
      data: {
        status: 'processing',
      },
    });
    
    // Transition to researching
    await transitionJob(submissionId, 'classifying', 'researching', config.name);
  } catch (error) {
    console.error(`Classifier failed for ${submissionId}:`, error);
    // Fallback: use safe defaults so pipeline can continue
    await job.updateData({
      ...job.data,
      classification: {
        issueType: 'policy',
        jurisdictionLevel: 'federal',
        severity: 'medium',
        confidence: 0.5,
        issueCategories: ['other'],
        summary: 'Classification failed — using defaults',
      },
    });
    await transitionJob(submissionId, 'classifying', 'researching', config.name);
  }
}

export const classifierWorker = new Worker(
  config.queue,
  processJob,
  {
    connection: createRedisConnection(),
    concurrency: 5,
  },
);

classifierWorker.on('failed', (job, err) => {
  console.error(`Classifier failed for job ${job?.id}:`, err.message);
});
```
