// eCFR API client for federal regulation lookup (SUBM-05)
// https://www.ecfr.gov/api/versioner/v1/full/{date}/title-{title}.xml

export interface ECFRResult {
  title: string;        // e.g., "42 CFR § 483.25"
  heading: string;      // e.g., "Quality of care"
  text: string;         // Regulation text excerpt
  url: string;          // Direct link to eCFR
  source: 'ecfr';
}

export async function searchECFR(query: string, issueCategories: string[]): Promise<ECFRResult[]> {
  const results: ECFRResult[] = [];
  
  try {
    // eCFR search API
    const searchUrl = `https://www.ecfr.gov/api/search/v1/results?query=${encodeURIComponent(query)}&per_page=5`;
    const response = await fetch(searchUrl);
    
    if (!response.ok) {
      console.error('eCFR search error:', response.status);
      return [];
    }
    
    const data = await response.json();
    
    for (const result of data?.results || []) {
      results.push({
        title: result.hierarchy_headings?.title || result.headings?.title || '',
        heading: result.hierarchy_headings?.section || result.headings?.section || '',
        text: result.full_text_excerpt || result.snippet || '',
        url: `https://www.ecfr.gov${result.hierarchy?.section_url || ''}`,
        source: 'ecfr',
      });
    }
  } catch (error) {
    console.error('eCFR API error:', error);
  }
  
  return results;
}

// Verify a specific eCFR citation exists (SUBM-06)
export async function verifyECFRCitation(citation: string): Promise<boolean> {
  try {
    // Parse CFR citation format: "XX CFR § YYY.ZZ"
    const match = citation.match(/(\d+)\s*CFR\s*§?\s*([\d.]+)/i);
    if (!match) return false;
    
    const [, title, section] = match;
    const url = `https://www.ecfr.gov/api/versioner/v1/full/current/title-${title}.json?section=${section}`;
    const response = await fetch(url, { method: 'HEAD' });
    
    return response.ok;
  } catch {
    return false;
  }
}

// CourtListener API client for case law lookup (SUBM-05)
// https://www.courtlistener.com/api/rest/v4/

export interface CourtListenerResult {
  caseName: string;     // e.g., "Brown v. Board of Education"
  citation: string;     // e.g., "347 U.S. 483 (1954)"
  court: string;        // e.g., "Supreme Court of the United States"
  dateDecided: string;
  snippet: string;      // Relevant excerpt
  url: string;
  source: 'courtlistener';
}

export async function searchCourtListener(query: string): Promise<CourtListenerResult[]> {
  const apiKey = process.env.COURTLISTENER_API_KEY;
  const results: CourtListenerResult[] = [];
  
  try {
    const headers: Record<string, string> = { 'Content-Type': 'application/json' };
    if (apiKey) headers['Authorization'] = `Token ${apiKey}`;
    
    const searchUrl = `https://www.courtlistener.com/api/rest/v4/search/?q=${encodeURIComponent(query)}&type=o&order_by=score+desc&page_size=5`;
    const response = await fetch(searchUrl, { headers });
    
    if (!response.ok) {
      console.error('CourtListener search error:', response.status);
      return [];
    }
    
    const data = await response.json();
    
    for (const result of data?.results || []) {
      results.push({
        caseName: result.caseName || '',
        citation: result.citation?.[0] || '',
        court: result.court || '',
        dateDecided: result.dateFiled || '',
        snippet: result.snippet || '',
        url: `https://www.courtlistener.com${result.absolute_url || ''}`,
        source: 'courtlistener',
      });
    }
  } catch (error) {
    console.error('CourtListener API error:', error);
  }
  
  return results;
}

// Verify a specific case citation exists (SUBM-06)
export async function verifyCourtListenerCitation(caseName: string): Promise<boolean> {
  try {
    const apiKey = process.env.COURTLISTENER_API_KEY;
    const headers: Record<string, string> = {};
    if (apiKey) headers['Authorization'] = `Token ${apiKey}`;
    
    const url = `https://www.courtlistener.com/api/rest/v4/search/?q=${encodeURIComponent(caseName)}&type=o&page_size=1`;
    const response = await fetch(url, { headers });
    
    if (!response.ok) return false;
    const data = await response.json();
    return (data?.results?.length || 0) > 0;
  } catch {
    return false;
  }
}

import { verifyECFRCitation } from './ecfr.js';
import { verifyCourtListenerCitation } from './courtlistener.js';

export interface Citation {
  text: string;           // The citation string, e.g., "42 CFR § 483.25"
  source: 'ecfr' | 'courtlistener' | 'state_cache';
  context: string;        // How it's used in the letter
}

export interface VerificationResult {
  citation: Citation;
  status: 'verified' | 'partial' | 'unverified';
  details?: string;
}

export interface VerificationSummary {
  results: VerificationResult[];
  verified: Citation[];
  unverified: Citation[];
  allFailed: boolean;     // If true, flag for human review (SUBM-08)
}

// Verify all citations in parallel (SUBM-06)
export async function verifyCitations(citations: Citation[]): Promise<VerificationSummary> {
  const results = await Promise.all(
    citations.map(async (citation): Promise<VerificationResult> => {
      try {
        switch (citation.source) {
          case 'ecfr': {
            const verified = await verifyECFRCitation(citation.text);
            return { citation, status: verified ? 'verified' : 'unverified' };
          }
          case 'courtlistener': {
            const verified = await verifyCourtListenerCitation(citation.text);
            return { citation, status: verified ? 'verified' : 'unverified' };
          }
          case 'state_cache': {
            // State cache entries are pre-verified (curated)
            return { citation, status: 'verified', details: 'Pre-verified curated cache' };
          }
          default:
            return { citation, status: 'unverified', details: 'Unknown source' };
        }
      } catch (error) {
        return { citation, status: 'unverified', details: `Verification error: ${error}` };
      }
    }),
  );
  
  const verified = results.filter(r => r.status === 'verified').map(r => r.citation);
  const unverified = results.filter(r => r.status === 'unverified').map(r => r.citation);
  
  return {
    results,
    verified,
    unverified,
    allFailed: verified.length === 0 && citations.length > 0,
  };
}

import { Worker, Job } from 'bullmq';
import { createRedisConnection } from '../engine/connection.js';
import { getAgentConfig } from '../engine/config.js';
import { transitionJob } from '../engine/state-machine.js';
import { callWithLogging } from '../lib/anthropic.js';
import { searchECFR } from '../lib/legal/ecfr.js';
import { searchCourtListener } from '../lib/legal/courtlistener.js';
import { searchStateCache } from '../lib/legal/state-cache.js';
import { verifyCitations, type Citation } from '../lib/legal/citation-verifier.js';

const config = getAgentConfig('researcher');

const RESEARCHER_SYSTEM_PROMPT = `You are a legal researcher for CivicState, a civic letter-writing platform. Your job is to find relevant regulations, case law, and statutes that support a citizen's civic concern.

You will be given:
1. The citizen's issue description and desired outcome
2. Classification results (issue type, jurisdiction, categories)
3. Search results from legal databases (eCFR, CourtListener, state statutes)

Your task:
1. Analyze the search results and select the most relevant regulations and cases
2. Synthesize a research brief explaining how these laws apply to the citizen's issue
3. Format citations for use in a formal letter

Respond with ONLY valid JSON:
{
  "researchBrief": "2-3 paragraph analysis of applicable regulations and how they support the citizen's position",
  "citations": [
    {
      "text": "The citation string (e.g., '42 CFR § 483.25')",
      "source": "ecfr" | "courtlistener" | "state_cache",
      "context": "How this citation supports the citizen's position (1-2 sentences)"
    }
  ],
  "recommendedApproach": "Brief guidance for the letter drafter on how to frame the argument",
  "caveat": "Any important limitations or caveats about the research"
}

IMPORTANT:
- Only cite regulations that appear in the provided search results
- Do not fabricate or hallucinate citations
- If search results are limited, acknowledge gaps honestly
- Include both federal and state regulations when applicable`;

async function processJob(job: Job): Promise<void> {
  const { submissionId, issueDescription, desiredOutcome, zipCode, classification } = job.data;
  const state = classification?.jurisdictionLevel === 'state' || classification?.jurisdictionLevel === 'multiple'
    ? zipCode.substring(0, 2) // Will be resolved properly via lookup
    : '';
  
  try {
    // Step 1: Search legal databases in parallel
    const searchQuery = `${issueDescription} ${classification?.issueCategories?.join(' ') || ''}`;
    const categories = classification?.issueCategories || ['other'];
    
    const [ecfrResults, courtResults, stateResults] = await Promise.all([
      searchECFR(searchQuery, categories),
      searchCourtListener(searchQuery),
      state ? searchStateCache(state, categories) : Promise.resolve([]),
    ]);
    
    // Step 2: Synthesize research via Sonnet 4.6
    const searchContext = JSON.stringify({
      ecfr: ecfrResults.slice(0, 5),
      courtlistener: courtResults.slice(0, 5),
      stateStatutes: stateResults.slice(0, 5),
    });
    
    const userMessage = `Issue: ${issueDescription}
Desired Outcome: ${desiredOutcome}
Classification: ${JSON.stringify(classification)}
ZIP Code: ${zipCode}

Legal Database Search Results:
${searchContext}`;
    
    const llmResult = await callWithLogging(
      'researcher',
      submissionId,
      'research',
      RESEARCHER_SYSTEM_PROMPT,
      userMessage,
    );
    
    const research = JSON.parse(llmResult.text);
    
    // Step 3: Verify all citations (SUBM-06)
    const citations: Citation[] = (research.citations || []).map((c: any) => ({
      text: c.text,
      source: c.source,
      context: c.context,
    }));
    
    const verification = await verifyCitations(citations);
    
    // Step 4: Handle verification results (SUBM-07, SUBM-08)
    if (verification.allFailed && citations.length > 0) {
      // All citations failed — flag for human review (SUBM-08)
      const { prisma } = await import('shared');
      await prisma.submission.update({
        where: { id: submissionId },
        data: { status: 'flagged' },
      });
      
      // Still transition to drafting but with flag
      await job.updateData({
        ...job.data,
        research: {
          ...research,
          citations: [],
          verificationStatus: 'all_failed',
          flaggedForReview: true,
        },
      });
    } else {
      // Strip unverified citations (SUBM-07)
      await job.updateData({
        ...job.data,
        research: {
          ...research,
          citations: verification.verified.map(c => ({
            text: c.text,
            source: c.source,
            context: c.context,
          })),
          strippedCitations: verification.unverified.map(c => c.text),
          verificationStatus: verification.unverified.length > 0 ? 'partial' : 'full',
        },
      });
    }
    
    // Transition to drafting
    await transitionJob(submissionId, 'researching', 'drafting', config.name);
  } catch (error) {
    console.error(`Researcher failed for ${submissionId}:`, error);
    // Transition to drafting with empty research (letter will use generic approach)
    await job.updateData({
      ...job.data,
      research: {
        researchBrief: 'Research unavailable — letter will use general civic advocacy approach.',
        citations: [],
        verificationStatus: 'failed',
      },
    });
    await transitionJob(submissionId, 'researching', 'drafting', config.name);
  }
}

export const researcherWorker = new Worker(
  config.queue,
  processJob,
  {
    connection: createRedisConnection(),
    concurrency: 5,
  },
);

researcherWorker.on('failed', (job, err) => {
  console.error(`Researcher failed for job ${job?.id}:`, err.message);
});
