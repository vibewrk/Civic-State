import { Worker, Job } from 'bullmq';
import { createRedisConnection } from '../engine/connection.js';
import { getAgentConfig } from '../engine/config.js';
import { transitionJob } from '../engine/state-machine.js';
import { callWithLogging } from '../lib/anthropic.js';

const config = getAgentConfig('drafter');

const AI_DISCLOSURE = `[AI Disclosure: This letter was drafted with AI assistance through CivicState.com. The research citations have been programmatically verified against government databases. This letter represents the views of the constituent, not of CivicState or any AI system.]`;

const LEGAL_DISCLAIMER = `[Disclaimer: This letter constitutes constituent communication, not legal advice. CivicState is not a law firm and does not provide legal services.]`;

const CAN_SPAM_FOOTER = `---
This email was sent by CivicState on behalf of a constituent.
CivicState, Inc. | 123 Main Street, Suite 100 | San Francisco, CA 94105
To opt out of future correspondence from CivicState: [OPT_OUT_URL]
This is a one-time transactional communication sent at the constituent's request.`;

const DRAFTER_SYSTEM_PROMPT = `You are a professional letter drafter for CivicState, a civic advocacy platform. You draft formal letters from citizens to government officials.

You will receive:
1. The citizen's issue description and desired outcome
2. Research findings with verified citations
3. Target official information (name, title, jurisdiction)
4. Whether the citizen is anonymous

Your letter MUST:
1. Be professional, respectful, and persuasive
2. Reference specific regulations and citations provided in the research
3. Clearly state the desired outcome / call to action
4. Be non-partisan in tone (facts and law, not political rhetoric)
5. Be 400-800 words
6. Follow standard formal letter format

You MUST NOT:
- Add any citations not provided in the research (they've been verified)
- Make legal claims or give legal advice
- Use inflammatory or threatening language
- Reference the citizen's name if they are anonymous

Respond with ONLY valid JSON:
{
  "salutation": "Dear [Title] [Name],",
  "body": "The letter body with paragraph breaks as \\n\\n",
  "closing": "Respectfully,",
  "signature": "The citizen's name or anonymous designation",
  "citationsUsed": ["list of citation strings used in the letter"],
  "wordCount": 500
}`;

async function processJob(job: Job): Promise<void> {
  const { submissionId, issueDescription, desiredOutcome, zipCode, isAnonymous, classification, research } = job.data;
  
  try {
    // Look up target officials for this submission
    const { prisma } = await import('shared');
    const submission = await prisma.submission.findUnique({
      where: { id: submissionId },
      include: { user: true },
    });
    
    // Get officials for this ZIP code from cache
    const officials = await prisma.official.findMany({
      where: { optedOut: false },
      take: 10,
    });
    
    // Create campaign record
    const campaign = await prisma.campaign.create({
      data: {
        submissionId,
        userId: submission?.userId || '',
        status: 'draft',
        pricingTier: 'single', // Will be updated at payment
        officialCount: officials.length,
      },
    });
    
    // Draft a letter for each official
    for (const official of officials) {
      const userMessage = `Issue: ${issueDescription}
Desired Outcome: ${desiredOutcome}
Classification: ${JSON.stringify(classification)}
Research: ${JSON.stringify(research)}

Target Official:
- Name: ${official.name}
- Title: ${official.title}
- Jurisdiction: ${official.jurisdiction}
- Level: ${official.level}

Citizen is anonymous: ${isAnonymous}
${!isAnonymous && submission?.user?.fullName ? `Citizen name: ${submission.user.fullName}` : ''}
ZIP Code: ${zipCode}`;
      
      const llmResult = await callWithLogging(
        'drafter',
        submissionId,
        'draft_letter',
        DRAFTER_SYSTEM_PROMPT,
        userMessage,
      );
      
      const draft = JSON.parse(llmResult.text);
      
      // Construct full letter with compliance elements
      const fullLetter = [
        draft.salutation,
        '',
        draft.body,
        '',
        AI_DISCLOSURE,
        '',
        LEGAL_DISCLAIMER,
        '',
        draft.closing,
        isAnonymous ? `A Concerned Constituent of ${zipCode}` : draft.signature,
        '',
        CAN_SPAM_FOOTER,
      ].join('\n');
      
      // Create letter record
      await prisma.letter.create({
        data: {
          campaignId: campaign.id,
          officialId: official.id,
          content: fullLetter, // TODO: Encrypt via CryptoService
          status: 'draft',
          aiDisclosure: AI_DISCLOSURE,
        },
      });
    }
    
    // Transition to payment_pending
    await transitionJob(submissionId, 'drafting', 'payment_pending', config.name);
  } catch (error) {
    console.error(`Drafter failed for ${submissionId}:`, error);
    await transitionJob(submissionId, 'drafting', 'failed', config.name);
  }
}

export const drafterWorker = new Worker(
  config.queue,
  processJob,
  {
    connection: createRedisConnection(),
    concurrency: 5,
  },
);

drafterWorker.on('failed', (job, err) => {
  console.error(`Drafter failed for job ${job?.id}:`, err.message);
});

// Poll research status (for the 1-3 minute loading screen)
router.get('/:id/research', async (req, res) => {
  const { prisma } = await import('shared');
  const { id } = req.params;
  
  const job = await prisma.job.findFirst({
    where: { submissionId: id },
    orderBy: { createdAt: 'desc' },
  });
  
  if (!job) {
    return res.status(404).json({ error: { code: 'NOT_FOUND', message: 'Submission not found' } });
  }
  
  // Map job status to research progress
  const statusMap: Record<string, string> = {
    submitted: 'queued',
    classifying: 'classifying_issue',
    researching: 'researching_regulations',
    drafting: 'drafting_letters',
    payment_pending: 'ready',
    paid: 'ready',
    delivering: 'ready',
    delivered: 'ready',
    failed: 'failed',
  };
  
  res.json({
    submissionId: id,
    status: statusMap[job.status] || job.status,
    jobStatus: job.status,
    createdAt: job.createdAt,
    // Research results included when available
    research: job.result || null,
  });
});

// Letter preview (collapsible cards per official)
router.get('/:id/preview', async (req, res) => {
  const { prisma } = await import('shared');
  const { id } = req.params;
  
  const campaign = await prisma.campaign.findFirst({
    where: { submissionId: id },
    include: {
      letters: {
        include: { official: true },
        where: { deletedAt: null },
      },
    },
  });
  
  if (!campaign) {
    return res.status(404).json({ error: { code: 'NOT_FOUND', message: 'No letters found for this submission' } });
  }
  
  const previews = campaign.letters.map(letter => ({
    letterId: letter.id,
    official: {
      name: letter.official.name,
      title: letter.official.title,
      jurisdiction: letter.official.jurisdiction,
      level: letter.official.level,
    },
    content: letter.content,
    status: letter.status,
    aiDisclosure: letter.aiDisclosure,
  }));
  
  res.json({
    submissionId: id,
    campaignId: campaign.id,
    officialCount: previews.length,
    letters: previews,
    disclaimer: 'This letter constitutes constituent communication, not legal advice.',
  });
});
