```typescript
const API_URL = process.env.NEXT_PUBLIC_API_URL || 'http://localhost:3001';

export interface SubmissionRequest {
  issueDescription: string;
  desiredOutcome: string;
  zipCode: string;
  isAnonymous: boolean;
  fullName?: string;
}

export interface SubmissionResponse {
  id: string;
  jobId: string;
  status: string;
}

export interface ResearchStatus {
  submissionId: string;
  status: string;
  jobStatus: string;
  research: any;
}

export interface LetterPreview {
  letterId: string;
  official: {
    name: string;
    title: string;
    jurisdiction: string;
    level: string;
  };
  content: string;
  status: string;
  aiDisclosure: string;
}

export interface PreviewResponse {
  submissionId: string;
  campaignId: string;
  officialCount: number;
  letters: LetterPreview[];
  disclaimer: string;
}

export interface OfficialInfo {
  name: string;
  title: string;
  email: string;
  jurisdiction: string;
  level: string;
  party?: string;
}

export interface OfficialsResponse {
  officials: OfficialInfo[];
  coverage: {
    federal: { found: number; expected: number };
    state: { found: number; expected: number };
    local: { found: number; expected: number };
  };
  confidenceLabel: string;
}

export async function createSubmission(data: SubmissionRequest): Promise<SubmissionResponse> {
  const res = await fetch(`${API_URL}/api/submissions`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(data),
  });
  if (!res.ok) {
    const error = await res.json();
    throw new Error(error.error?.message || 'Submission failed');
  }
  return res.json();
}

export async function getResearchStatus(submissionId: string): Promise<ResearchStatus> {
  const res = await fetch(`${API_URL}/api/submissions/${submissionId}/research`);
  if (!res.ok) throw new Error('Failed to get research status');
  return res.json();
}

export async function getLetterPreviews(submissionId: string): Promise<PreviewResponse> {
  const res = await fetch(`${API_URL}/api/submissions/${submissionId}/preview`);
  if (!res.ok) throw new Error('Failed to get letter previews');
  return res.json();
}

export async function lookupOfficials(zipCode: string): Promise<OfficialsResponse> {
  const res = await fetch(`${API_URL}/api/officials?zipCode=${zipCode}`);
  if (!res.ok) throw new Error('Failed to look up officials');
  return res.json();
}
```
