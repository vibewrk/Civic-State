```typescript
// Congress.gov API client for federal official lookup (OFCL-01)
// Federal: 2 senators (by state) + 1 representative (by district) = 3 expected

export interface FederalOfficial {
  name: string;
  title: string;
  state: string;
  district?: string;
  party: string;
  email?: string;
  phone?: string;
  url?: string;
}

// ZIP → state + congressional district via census geocoder
async function zipToDistrict(zipCode: string): Promise<{ state: string; district: string } | null> {
  try {
    const url = `https://geocoding.geo.census.gov/geocoder/geographies/onelineaddress?address=${zipCode}&benchmark=Public_AR_Current&vintage=Current_Current&format=json`;
    const response = await fetch(url);
    const data = await response.json();
    
    const geographies = data?.result?.addressMatches?.[0]?.geographies;
    const congressDistricts = geographies?.['118th Congressional Districts'];
    const states = geographies?.['States'];
    
    if (!congressDistricts?.length || !states?.length) return null;
    
    return {
      state: states[0].STUSAB,
      district: congressDistricts[0].CD118,
    };
  } catch {
    return null;
  }
}

// Look up members of congress via congress.gov API
export async function lookupFederalOfficials(zipCode: string): Promise<FederalOfficial[]> {
  const location = await zipToDistrict(zipCode);
  if (!location) return [];
  
  const apiKey = process.env.CONGRESS_GOV_API_KEY;
  const headers: Record<string, string> = {};
  if (apiKey) headers['X-API-Key'] = apiKey;
  
  const officials: FederalOfficial[] = [];
  
  try {
    // Fetch senators (2 per state)
    const senatorsUrl = `https://api.congress.gov/v3/member?stateCode=${location.state}&chamber=Senate&currentMember=true&limit=2&format=json`;
    const senRes = await fetch(senatorsUrl, { headers });
    if (senRes.ok) {
      const senData = await senRes.json();
      for (const m of senData?.members || []) {
        officials.push({
          name: m.name || `${m.firstName} ${m.lastName}`,
          title: 'U.S. Senator',
          state: location.state,
          party: m.partyName || m.party || '',
          url: m.url,
        });
      }
    }
    
    // Fetch representative (1 per district)
    const repUrl = `https://api.congress.gov/v3/member?stateCode=${location.state}&district=${location.district}&chamber=House&currentMember=true&limit=1&format=json`;
    const repRes = await fetch(repUrl, { headers });
    if (repRes.ok) {
      const repData = await repRes.json();
      for (const m of repData?.members || []) {
        officials.push({
          name: m.name || `${m.firstName} ${m.lastName}`,
          title: 'U.S. Representative',
          state: location.state,
          district: location.district,
          party: m.partyName || m.party || '',
          url: m.url,
        });
      }
    }
  } catch (error) {
    console.error('Congress.gov API error:', error);
  }
  
  return officials;
}
```

```typescript
// OpenStates v3 API client for state legislator lookup (OFCL-02)
// State: typically 1 state senator + 1 state representative = 2 expected

export interface StateOfficial {
  name: string;
  title: string;
  state: string;
  district: string;
  party: string;
  email?: string;
  phone?: string;
  chamber: string;
}

const OPENSTATES_URL = 'https://v3.openstates.org/graphql';

// OpenStates v3 uses GraphQL — query people by geographic coordinates from ZIP
export async function lookupStateOfficials(zipCode: string, state: string): Promise<StateOfficial[]> {
  const apiKey = process.env.OPENSTATES_API_KEY;
  if (!apiKey) {
    console.warn('OPENSTATES_API_KEY not set — skipping state official lookup');
    return [];
  }
  
  const officials: StateOfficial[] = [];
  
  try {
    // Use the people search endpoint with location
    const url = `https://v3.openstates.org/people?jurisdiction=${state.toLowerCase()}&include=other_identifiers&apikey=${apiKey}`;
    const response = await fetch(url);
    
    if (!response.ok) {
      console.error('OpenStates API error:', response.status);
      return [];
    }
    
    const data = await response.json();
    
    for (const person of data?.results || []) {
      if (!person.current_role) continue;
      
      officials.push({
        name: person.name,
        title: person.current_role.title || (person.current_role.chamber === 'upper' ? 'State Senator' : 'State Representative'),
        state: state,
        district: person.current_role.district || '',
        party: person.party || '',
        email: person.email || '',
        chamber: person.current_role.chamber || '',
      });
    }
  } catch (error) {
    console.error('OpenStates API error:', error);
  }
  
  return officials;
}
```

```typescript
import { lookupFederalOfficials } from './congress.js';
import { lookupStateOfficials } from './openstates.js';
import { lookupLocalOfficials } from './cicero.js';

export interface OfficialResult {
  name: string;
  title: string;
  email: string;
  jurisdiction: string;
  level: 'federal' | 'state' | 'local';
  district?: string;
  state?: string;
  party?: string;
  phone?: string;
  sourceApi: string;
}

export interface LookupResult {
  officials: OfficialResult[];
  coverage: {
    federal: { found: number; expected: number };
    state: { found: number; expected: number };
    local: { found: number; expected: number };
  };
  confidenceLabel: string;
}

export async function lookupOfficials(zipCode: string): Promise<LookupResult> {
  // Run all three lookups in parallel
  const [federal, state, local] = await Promise.all([
    lookupFederalOfficials(zipCode),
    lookupStateOfficials(zipCode, ''), // State will be derived from ZIP in congress lookup
    lookupLocalOfficials(zipCode),
  ]);
  
  const officials: OfficialResult[] = [];
  
  for (const f of federal) {
    officials.push({
      name: f.name,
      title: f.title,
      email: '', // Congress.gov doesn't provide direct email — use contact forms
      jurisdiction: 'U.S. Congress',
      level: 'federal',
      district: f.district,
      state: f.state,
      party: f.party,
      sourceApi: 'congress.gov',
    });
  }
  
  for (const s of state) {
    officials.push({
      name: s.name,
      title: s.title,
      email: s.email || '',
      jurisdiction: `${s.state} State Legislature`,
      level: 'state',
      district: s.district,
      state: s.state,
      party: s.party,
      sourceApi: 'openstates',
    });
  }
  
  for (const l of local) {
    officials.push({
      name: l.name,
      title: l.title,
      email: l.email || '',
      jurisdiction: l.jurisdiction,
      level: 'local',
      district: l.district,
      party: l.party,
      phone: l.phone,
      sourceApi: 'cicero',
    });
  }
  
  const coverage = {
    federal: { found: federal.length, expected: 3 }, // 2 senators + 1 rep
    state: { found: state.length, expected: 2 },     // 1 state senator + 1 state rep
    local: { found: local.length, expected: 5 },     // varies — 5 is estimate
  };
  
  const totalFound = officials.length;
  const totalExpected = coverage.federal.expected + coverage.state.expected + coverage.local.expected;
  const confidenceLabel = `We found ${totalFound} of approximately ${totalExpected} officials for your area`;
  
  return { officials, coverage, confidenceLabel };
}

// Cache officials in DB and enforce opt-out (OFCL-04, OFCL-06)
export async function cacheAndFilterOfficials(lookupResult: LookupResult): Promise<LookupResult> {
  const { prisma } = await import('shared');
  
  const filtered: OfficialResult[] = [];
  
  for (const official of lookupResult.officials) {
    // Upsert official in cache
    const cached = await prisma.official.upsert({
      where: {
        email_level: { email: official.email || `${official.name.toLowerCase().replace(/\s/g, '.')}@placeholder`, level: official.level },
      },
      update: {
        name: official.name,
        title: official.title,
        jurisdiction: official.jurisdiction,
        district: official.district,
        state: official.state,
        party: official.party,
        phone: official.phone,
        sourceApi: official.sourceApi,
        lastVerifiedAt: new Date(),
      },
      create: {
        name: official.name,
        title: official.title,
        email: official.email || `${official.name.toLowerCase().replace(/\s/g, '.')}@placeholder`,
        jurisdiction: official.jurisdiction,
        level: official.level,
        district: official.district,
        state: official.state,
        party: official.party,
        phone: official.phone,
        sourceApi: official.sourceApi,
        lastVerifiedAt: new Date(),
      },
    });
    
    // OFCL-06: Enforce opt-out — suppressed across ALL users
    if (!cached.optedOut) {
      filtered.push({ ...official, email: cached.email });
    }
  }
  
  return {
    ...lookupResult,
    officials: filtered,
    confidenceLabel: filtered.length < lookupResult.officials.length
      ? `${lookupResult.confidenceLabel} (${lookupResult.officials.length - filtered.length} opted out)`
      : lookupResult.confidenceLabel,
  };
}
```

```typescript
import { Router } from 'express';
import { z } from 'zod';
import { lookupOfficials, cacheAndFilterOfficials } from '../lib/officials/lookup.js';
import rateLimit from 'express-rate-limit';

const router: Router = Router();

// Rate limit: 100 requests per 15 min per IP (SUBM-10, consensus: api-design)
const officialsLimiter = rateLimit({
  windowMs: 15 * 60 * 1000,
  max: 100,
  standardHeaders: true,
  legacyHeaders: false,
  message: { error: { code: 'RATE_LIMITED', message: 'Too many requests. Try again in 15 minutes.' } },
});

const zipSchema = z.object({
  zipCode: z.string().regex(/^\d{5}(-\d{4})?$/, 'Invalid ZIP code'),
});

// GET /api/officials?zipCode=12345 — public, rate-limited, cached
router.get('/', officialsLimiter, async (req, res) => {
  try {
    const { zipCode } = zipSchema.parse({ zipCode: req.query.zipCode });
    
    const result = await lookupOfficials(zipCode);
    const filtered = await cacheAndFilterOfficials(result);
    
    res.json({
      officials: filtered.officials,
      coverage: filtered.coverage,
      confidenceLabel: filtered.confidenceLabel,
    });
  } catch (error) {
    if (error instanceof z.ZodError) {
      return res.status(400).json({ error: { code: 'VALIDATION_ERROR', message: 'Invalid ZIP code format' } });
    }
    console.error('Officials lookup error:', error);
    res.status(500).json({ error: { code: 'LOOKUP_FAILED', message: 'Officials lookup failed' } });
  }
});

export default router;
```
