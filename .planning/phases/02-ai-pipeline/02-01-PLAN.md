Output: Working moderation pipeline that auto-blocks threats, flags risky content for human review, and passes everything else with full audit logging.

```typescript
import Anthropic from '@anthropic-ai/sdk';

export interface ModerationResult {
  tier: 'block' | 'flag' | 'pass';
  reason: string;
  confidence: number;
  details?: string;
}

// Keyword blocklist for immediate detection (Tier 1 — MODR-01)
const THREAT_PATTERNS = [
  /\b(kill|murder|assassinate|bomb|shoot|stab)\b.*\b(you|them|him|her|official|senator|representative|mayor|governor)\b/i,
  /\b(i('ll| will)|going to|gonna)\b.*\b(hurt|harm|attack|destroy)\b/i,
  /\bdeath\s+threat/i,
  /\bswat(ting)?\b/i,
];

function keywordCheck(text: string): ModerationResult | null {
  for (const pattern of THREAT_PATTERNS) {
    if (pattern.test(text)) {
      return { tier: 'block', reason: 'criminal_threat', confidence: 0.99, details: 'Keyword blocklist match' };
    }
  }
  return null;
}

// LLM-based classification (Tier 1 auto-block + Tier 2 flag — MODR-01, MODR-02, MODR-03)
const MODERATION_PROMPT = `You are a content moderation classifier for a civic letter-writing platform. Analyze the user's submission for safety issues.

IMPORTANT: This platform is NON-PARTISAN. Do NOT flag content based on political viewpoint, ideology, or party affiliation. Citizens have the right to express any political opinion to their elected officials.

Classify the content into one of these tiers:

TIER 1 - AUTO-BLOCK (confidence must be >95%):
- Direct threats of violence against any person
- Explicit harassment targeting individuals
- Hate speech targeting protected groups
- Criminal threats or incitement to violence

TIER 2 - FLAG FOR REVIEW (confidence must be >80%):
- Named misconduct allegations against specific officials (e.g., "Mayor X took bribes")
- Potentially defamatory statements (unproven accusations of crimes/corruption)
- Unverifiable claims presented as fact
- References to active litigation

TIER 3 - PASS:
- All other content, including strong political opinions, criticism of policies, complaints about government performance, demands for action

Respond with ONLY valid JSON:
{
  "tier": "block" | "flag" | "pass",
  "reason": "threat" | "harassment" | "hate_speech" | "criminal_threat" | "defamation_risk" | "misconduct_allegation" | "unverifiable_claim" | "clean",
  "confidence": 0.0-1.0,
  "details": "brief explanation"
}`;

let anthropicClient: Anthropic | null = null;

function getAnthropicClient(): Anthropic {
  if (!anthropicClient) {
    anthropicClient = new Anthropic();
  }
  return anthropicClient;
}

export async function moderateContent(
  issueDescription: string,
  desiredOutcome: string,
): Promise<ModerationResult> {
  const fullText = `${issueDescription}\n\nDesired Outcome: ${desiredOutcome}`;

  // Step 1: Keyword blocklist (fast, no API call)
  const keywordResult = keywordCheck(fullText);
  if (keywordResult) {
    return keywordResult;
  }

  // Step 2: LLM classification via Haiku 4.5
  try {
    const client = getAnthropicClient();
    const response = await client.messages.create({
      model: 'claude-haiku-4-5-20250401',
      max_tokens: 256,
      system: MODERATION_PROMPT,
      messages: [{ role: 'user', content: fullText }],
    });

    const text = response.content[0].type === 'text' ? response.content[0].text : '';
    const parsed = JSON.parse(text);

    // Enforce confidence thresholds (MODR-03)
    if (parsed.tier === 'block' && parsed.confidence < 0.95) {
      // Below block threshold, downgrade to flag
      return { tier: 'flag', reason: parsed.reason, confidence: parsed.confidence, details: parsed.details };
    }
    if (parsed.tier === 'flag' && parsed.confidence < 0.80) {
      // Below flag threshold, pass
      return { tier: 'pass', reason: 'clean', confidence: parsed.confidence, details: 'Below flag confidence threshold' };
    }

    return {
      tier: parsed.tier,
      reason: parsed.reason,
      confidence: parsed.confidence,
      details: parsed.details,
    };
  } catch (error) {
    // If moderation API fails, flag for human review (fail-safe)
    console.error('Moderation API error:', error);
    return { tier: 'flag', reason: 'unverifiable_claim', confidence: 0, details: 'Moderation API unavailable — flagged for manual review' };
  }
}
```
